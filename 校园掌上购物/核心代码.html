1. 登录模块

1.1 登录页面核心逻辑（views/login/index.vue）

<script setup lang="ts">
import { ref, reactive, toRefs, onMounted } from 'vue'
import { isUsername, isPassword } from '@/utils/validate'
import { useAuthStore } from '@/stores/auth'
import { useRouter, useRoute } from 'vue-router'
import { Session } from '@/utils/storage'
import menu from "@/utils/menu"

const authStore = useAuthStore()
const router = useRouter()
const route = useRoute()

const state = reactive({
    loading: false,
    loginData: { username: '', password: '', role: '' },
    tableName: ''
})

const menus = menu.list()
const { loading, loginData, tableName } = { ...toRefs(state) }

const showPassword = ref(false)
const formData = reactive({ captcha: '' })
const captchaCode = ref('')

// 生成验证码
const generateCaptcha = () => {
    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789'
    let captcha = ''
    for (let i = 0; i < 4; i++) {
        captcha += chars.charAt(Math.floor(Math.random() * chars.length))
    }
    return captcha
}

const refreshCaptcha = () => {
    captchaCode.value = generateCaptcha()
}

const handleSubmit = async () => {
    state.loading = true
    try {
        if (!state.loginData.username || !state.loginData.password) {
            notify("账号密码不能为空", { type: "error" })
            return
        }
        // 验证码校验
        if (captchaCode.value.toUpperCase() !== formData.captcha.toUpperCase()) {
            notify("验证码错误", { type: "error" })
            refreshCaptcha()
            return
        }
        // 根据角色获取表名
        for (let i = 0; i < menus.length; i++) {
            if (menus[i].roleName == state.loginData.role) {
                state.tableName = menus[i].tableName
            }
        }
        // 调用登录接口
        const res = await authStore.userLogin(state.loginData, `${state.tableName}/login`)
        Session.set("tableName", state.tableName)
        Session.set("role", state.loginData.role)
        Session.set("adminName", state.loginData.username)
        router.replace({ path: route.query?.redirect || '/' })
    } finally {
        state.loading = false
    }
}
</script>

1.2 认证状态管理（stores/auth.ts）

import { defineStore } from 'pinia'
import { GetToken } from '@/api/auth'
import { Session, Local } from '@/utils/storage'

export const Key = {
    accessTokenKey: 'accessToken',
    userInfoKey: 'userInfo',
}

export const useAuthStore = defineStore('auth', {
    state: () => {
        return {
            accessToken: Session.get(Key.accessTokenKey),
            users: Session.get(Key.userInfoKey)
        }
    },

    actions: {
        userLogin(loginData: any, url: string) {
            return new Promise((resolve, reject) => {
                GetToken(loginData, url).then((res: any) => {
                    this.accessToken = res.token
                    if (Session.get(Key.accessTokenKey) != null) {
                        Session.remove(Key.accessTokenKey)
                    }
                    Session.set(Key.accessTokenKey, res.token)
                    resolve(res)
                }).catch((error: Error) => {
                    reject(error)
                })
            })
        }
    }
})

function submitForm() {
    formRef.value?.validate((valid: boolean) => {
        if (!valid) return;
        state.loadding = true;
        if (state.type == 'cross') {

                url: `cart/save`,

}

10. 公共API调用

10.1 请求拦截器（utils/request.ts）

import axios from 'axios'
import type { AxiosInstance } from 'axios'
import { ElMessage } from 'element-plus'
import { useAuthStore } from '@/stores/auth'
import router from '@/router'

const request: AxiosInstance = axios.create({
    baseURL: import.meta.env.VITE_APP_BASE_API as string,
    timeout: 20000,
})

// 请求拦截器
request.interceptors.request.use((config) => {
    const authStore = useAuthStore()
    const accessToken = authStore.accessToken
    if (accessToken) {
        config.headers['Token'] = accessToken
    }
    return config
}, (error) => {
    return Promise.reject(error)
})

// 响应拦截器
request.interceptors.response.use(response => {
    const { data } = response
    if (data.code === 0) {
        return data
    } else if (data.code === 401) {
        ElMessage.error(data.msg)
        setTimeout(() => {
            router.push({ path: '/login' })
        }, 300)
    }
    ElMessage.error(data.msg)
    return Promise.reject(response)
}, error => {
    const { message } = error
    if (message.indexOf('timeout') != -1) {
        ElMessage.error('网络超时！')
    } else if (message == 'Network Error') {
        ElMessage.error('网络连接错误！')
    }
    return Promise.reject(error)
})

export default request

10.2 存储工具（utils/storage.ts）

// 浏览器永久缓存 (localStorage)
export const Local = {
    set(key: string, val: any) {
        window.localStorage.setItem(key, JSON.stringify(val || ''))
    },
    get(key: string) {
        let json = window.localStorage.getItem(key)
        return JSON.parse(json)
    },
    remove(key: string) {
        window.localStorage.removeItem(key)
    },
    clear() {
        window.localStorage.clear()
    },
}

// 浏览器临时缓存 (sessionStorage)
export const Session = {
    set(key: string, val: any) {
        window.sessionStorage.setItem(key, JSON.stringify(val || ''))
    },
    get(key: string) {
        let json = window.sessionStorage.getItem(key)
        return JSON.parse(json)
    },
    remove(key: string) {
        window.sessionStorage.removeItem(key)
    },
    clear() {
        window.sessionStorage.clear()
    },
}
